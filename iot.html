Embedded capture loop (C)
/* edge camera firmware — illustrative */
#include <ptm_hal.h>
#include <ptm_net.h>

static frame_buf_t fb;
static mq_client_t mq;

int ptm_capture_task(void *arg) {
    (void)arg;
    sensor_init(&fb, RES_1080P, FPS_25);

    while (!ptm_shutdown_requested()) {
        if (sensor_grab(&fb) != PTM_OK) {
            watchdog_kick();
            continue;
        }
        /* lightweight pre-filter on the MCU before
           shipping to the edge AI node */
        if (motion_delta(&fb) > MOTION_THRESHOLD) {
            det_t d;
            if (infer_local(&fb, &d) && d.conf >= 0.60f) {
                mq_pub_detection(&mq, &d);
            }
            ring_push(&evidence_ring, &fb);
        }
        telemetry_tick(&mq);   /* temp, uptime, tx rate */
        watchdog_kick();
        vTaskDelayUntil(&fb.ts, pdMS_TO_TICKS(40));
    }
    return 0;
}

Firmware keeps a rolling evidence ring so BEFORE frames survive the moment a violation is confirmed. Detections below the confidence threshold are published with a needs_review flag rather than being actioned.
